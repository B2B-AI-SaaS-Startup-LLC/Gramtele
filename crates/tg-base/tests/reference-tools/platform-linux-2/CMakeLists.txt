cmake_minimum_required(VERSION 3.25)
project(platform_linux_2_reference LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
find_package(Qt6 REQUIRED COMPONENTS Core)
find_library(NATIVE_QT_GUI Qt6Gui HINTS "${Qt6_DIR}/../.." REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/extract.py ${TELEGRAM_ROOT} ${CMAKE_CURRENT_BINARY_DIR} COMMAND_ERROR_IS_FATAL ANY)
add_executable(native_linux_2 main.cpp ${TELEGRAM_ROOT}/lib_base/base/assertion.cpp ${TELEGRAM_ROOT}/lib_base/base/integration.cpp ${TELEGRAM_ROOT}/lib_base/base/debug_log.cpp)
target_include_directories(native_linux_2 PRIVATE ${CMAKE_CURRENT_BINARY_DIR} ${TELEGRAM_ROOT}/lib_base ${TELEGRAM_ROOT}/lib_rpl ${TELEGRAM_ROOT}/lib_crl ${GSL_INCLUDE} ${RANGE_INCLUDE} ${XKB_INCLUDE})
target_compile_options(native_linux_2 PRIVATE -O1 -include ${CMAKE_CURRENT_SOURCE_DIR}/prefix.hpp -ffunction-sections -fdata-sections)
target_link_options(native_linux_2 PRIVATE -Wl,--gc-sections)
target_link_libraries(native_linux_2 PRIVATE Qt6::Core ${NATIVE_QT_GUI})
