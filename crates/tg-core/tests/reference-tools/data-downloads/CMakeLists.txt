cmake_minimum_required(VERSION 3.25)
project(downloads_reference LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
find_package(Qt6 REQUIRED COMPONENTS Core)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/extract.py ${TELEGRAM_ROOT} ${CMAKE_CURRENT_BINARY_DIR} COMMAND_ERROR_IS_FATAL ANY)
add_executable(native_downloads main.cpp ${CMAKE_CURRENT_BINARY_DIR}/codec.cpp ${TELEGRAM_ROOT}/lib_base/base/assertion.cpp ${TELEGRAM_ROOT}/lib_base/base/integration.cpp)
target_include_directories(native_downloads PRIVATE ${CMAKE_CURRENT_BINARY_DIR} ${SCHEME_INCLUDE} ${TELEGRAM_ROOT}/SourceFiles ${TELEGRAM_ROOT}/lib_base ${TELEGRAM_ROOT}/lib_crl ${TELEGRAM_ROOT}/lib_rpl ${TELEGRAM_ROOT}/lib_tl ${GSL_INCLUDE} ${RANGE_INCLUDE})
target_compile_options(native_downloads PRIVATE -O0 -include ${CMAKE_CURRENT_SOURCE_DIR}/prefix.hpp -ffunction-sections -fdata-sections)
target_link_options(native_downloads PRIVATE -Wl,--gc-sections)
target_link_libraries(native_downloads PRIVATE Qt6::Core)
