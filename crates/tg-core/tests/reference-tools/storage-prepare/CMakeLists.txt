cmake_minimum_required(VERSION 3.25)
project(storage_prepare_reference LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
find_package(Qt6 REQUIRED COMPONENTS Core)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/extract.py ${TELEGRAM_ROOT} ${CMAKE_CURRENT_BINARY_DIR} COMMAND_ERROR_IS_FATAL ANY)
add_executable(native_storage_prepare main.cpp)
target_include_directories(native_storage_prepare PRIVATE ${CMAKE_CURRENT_BINARY_DIR} ${TELEGRAM_ROOT}/lib_base ${GSL_INCLUDE} ${RANGE_INCLUDE})
target_compile_options(native_storage_prepare PRIVATE -O2)
target_link_libraries(native_storage_prepare PRIVATE Qt6::Core)
